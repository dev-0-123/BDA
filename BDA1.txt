// Switch to a new database (or create it if it doesn't exist)
use mydatabase

// Create a collection (or use an existing one)
db.createCollection("students")

**INSERT**
// Insert single document
db.students.insertOne({
    name: "Aarav Sharma",
    age: 22,
    city: "Mumbai"
})

// Insert multiple documents
db.students.insertMany([
    { name: "Priya Singh", age: 20, city: "Delhi" },
    { name: "Rahul Verma", age: 24, city: "Bangalore" },
    { name: "Ananya Iyer", age: 21, city: "Chennai" },
    { name: "Naina Joshi", age: 22, city: "Chennai" }
])

**QUERY**
// Find all documents in the collection
db.students.find()

// Find documents with a specific condition (e.g., city = "Mumbai")
db.students.find({ city: "Mumbai" })

// Find a single document
db.students.findOne({ name: "Priya Singh" })

**UPDATE**
// Update a single document (e.g., update age for Priya Singh)
db.students.updateOne(
    { name: "Priya Singh" },
    { $set: { age: 22 } }
)

// Update multiple documents (e.g., increment age by 1 for all students in Mumbai)
db.students.updateMany(
    { city: "Mumbai" },
    { $inc: { age: 1 } }
)

**DELETE**
// Delete a single document
db.students.deleteOne({ name: "Aarav Sharma" })

// Delete multiple documents
db.students.deleteMany({ city: "Chennai" })
